*!*	oldalias=ALIAS()
*!*	SELECT xm,SUM(1) FROM lstt GROUP BY xm having(SUM(1))>1 INTO CURSOR tc
*!*	SELECT tc
*!*	cexp=""
*!*	SCAN
*!*	    cexp=cexp+" "+xm
*!*	ENDSCAN

*!*	IF !EMPTY(oldalias)
*!*		SELECT (oldalias)
*!*	ENDIF

*!*	IF !EMPTY(cexp)
*!*	    MESSAGEBOX("系统内有以下重复姓名，请修改以姓名区别："+CHR(13)+CHR(10)+cexp,64,"提示")
*!*	    RETURN
*!*	ENDIF    


v_filename=GETFILE("EXCEL表:XLS,XLSX","打开文件","确定",0,"选择文件")

IF EMPTY(V_FILENAME)
    RETURN
ENDIF

IF !INLIST(UPPER(JUSTEXT(v_filename)),"XLS","XLSX")
    RETURN
ENDIF

*!*	TRY
	xlApp = CreateObject("Excel.Application")
	if error()=1733
	    messagebox('该功能需要Excel支持，请安装。',0,'系统提示')
	    return
	endif
	xlapp.StandardFontSize = "9"
	xlapp.SheetsInNewWorkbook = 1

	xlBook = xlApp.Workbooks.open(v_filename)

	xlSheet = xlBook.Worksheets(1)

	cexpr=""
	i=1
	DO WHILE .t.
	    fldname=ALLTRIM(CAST(xlSheet.cells(1,i).value as c(18)))
	    fldname=STRTRAN(fldname,CHR(13)+CHR(10),'')
	    fldname=STRTRAN(fldname,CHR(13),'')
	    fldname=STRTRAN(fldname,CHR(10),'')
	    fldname=STRTRAN(fldname,' ','')
	    IF EMPTY(fldname) OR ISNULL(fldname)
	        EXIT
	    ELSE
	        IF ALLTRIM(fldname)='姓名'
		        cexpr=cexpr+","+ALLTRIM(fldname)+" c(8)"
		    ELSE
		        cexpr=cexpr+","+ALLTRIM(fldname)+" c(30)"
	        ENDIF
	        i=i+1
	    ENDIF
	ENDDO
	cexpr="CREATE CURSOR lxls ("+SUBSTR(cexpr,2)+",id i autoinc,ryid i)"
*!*		cexpr="CREATE CURSOR lxls ("+cexpr+")"
	&cexpr

	fldcnt=i-1

	SET DECIMALS TO 2
	j=2
	DO WHILE .t.
	    cc=""
	    FOR i=1 TO fldcnt
	        fldname=ALLTRIM(CAST(xlSheet.cells(1,i).value as c(18)))
	        aa=xlSheet.cells(j,i).value
	        IF TYPE("aa")="N"
	            IF INT(aa)=aa
	                fldv=CAST(INT(aa) as c(20))
	            ELSE
	                fldv=CAST(aa as c(20))
	                fldv=LEFT(fldv,AT(".",fldv)+2)
	            ENDIF
	        ELSE
	            fldv=ALLTRIM(CAST(xlSheet.cells(j,i).value as c(254)))
	        ENDIF
	        IF ISNULL(fldv)
	            fldv=""
	        ELSE
	            fldv=ALLTRIM(fldv)
	        ENDIF
	        cc=cc+fldv
	        STORE fldv TO &fldname
	    ENDFOR
	    IF !EMPTY(cc)
	        INSERT INTO lxls FROM MEMVAR
	    ELSE
	        EXIT
	    ENDIF
	    i=1
	    j=j+1
	ENDDO

	IF FIELD(1,"lxls")<>'单位编码' OR FIELD(2,"lxls")<>'个人编码'
	    MESSAGEBOX("EXCEL表的第1列标题必须“单位编码”,第2列标题必须“个人编码”。",64,"提示")
	    USE IN lxls
	    RETURN
	ENDIF


	DO FORM forms\drbrow WITH V_FILENAME TO v_success

*!*		IF v_success
*!*			thisform.showdata()
*!*		ENDIF

	xlBook.close(.f.)
	RELEASE xlapp
